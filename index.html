function doGet(e) {
  if (e && e.parameter && e.parameter.action === 'getOrders') {
    var orders = getAllOrdersFromSheet();
    return ContentService.createTextOutput(JSON.stringify({ ok: true, orders: orders }))
      .setMimeType(ContentService.MimeType.JSON);
  }
  return HtmlService.createHtmlOutputFromFile('index')
    .setTitle('축제 바리스타 카페')
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL)
    .addMetaTag('viewport', 'width=device-width, initial-scale=1');
}

function getOrCreateCafeSheet() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('카페주문내역') || ss.insertSheet('카페주문내역');
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(['주문ID','주문번호','주문시각','수령방식','배달장소','주문자','결제수단','입금자명','주문내역','총금액','요청사항','상태','상세JSON']);
  }
  return sheet;
}

function createOrderInSheet(order) {
  var sheet = getOrCreateCafeSheet();
  var nextNum = sheet.getLastRow();
  order.orderNo = '#' + ('000' + nextNum).slice(-3);
  sheet.appendRow([
    String(order.id),
    order.orderNo,
    order.createdAt,
    order.orderType,
    order.location,
    order.customerName,
    order.paymentMethod,
    order.depositorName,
    order.itemsSummary,
    order.totalPrice,
    order.memo,
    order.status,
    JSON.stringify(order.items)
  ]);
  return getAllOrdersFromSheet();
}

function getAllOrdersFromSheet() {
  var sheet = getOrCreateCafeSheet();
  var lastRow = sheet.getLastRow();
  if (lastRow <= 1) return [];
  var values = sheet.getRange(2, 1, lastRow - 1, 13).getValues();
  var list = [];
  for (var i = values.length - 1; i >= 0; i--) {
    var r = values[i];
    var parsedItems = [];
    try { parsedItems = JSON.parse(r[12] || '[]'); } catch(err) {}
    list.push({
      id: Number(r[0]) || (i + 1),
      orderNo: String(r[1]),
      createdAt: String(r[2]),
      orderType: String(r[3]),
      location: String(r[4]),
      customerName: String(r[5]),
      paymentMethod: String(r[6]),
      depositorName: String(r[7]),
      itemsSummary: String(r[8]),
      totalPrice: Number(r[9]) || 0,
      memo: String(r[10]),
      status: String(r[11]) || '대기중',
      items: parsedItems
    });
  }
  return list;
}

function toggleOrderStatusInSheet(orderId, newStatus) {
  var sheet = getOrCreateCafeSheet();
  var lastRow = sheet.getLastRow();
  if (lastRow <= 1) return [];
  var ids = sheet.getRange(2, 1, lastRow - 1, 1).getValues();
  for (var i = 0; i < ids.length; i++) {
    if (String(ids[i][0]) === String(orderId)) {
      sheet.getRange(i + 2, 12).setValue(newStatus);
      break;
    }
  }
  return getAllOrdersFromSheet();
}

function doPost(e) {
  var data = JSON.parse(e.postData.contents);
  if (data.action === 'create') {
    createOrderInSheet(data.order);
  } else if (data.action === 'status') {
    toggleOrderStatusInSheet(data.orderId, data.status);
  }
  return ContentService.createTextOutput(JSON.stringify({ ok: true })).setMimeType(ContentService.MimeType.JSON);
}
